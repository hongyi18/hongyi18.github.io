<script>
  (function () {
    var entries = Array.prototype.slice.call(document.querySelectorAll('.activity-entry'));
    var buttons = Array.prototype.slice.call(document.querySelectorAll('[data-activity-filter]'));
    var years = Array.prototype.slice.call(document.querySelectorAll('[data-year-section]'));
    var yearActionButtons = Array.prototype.slice.call(document.querySelectorAll('[data-year-action]'));
    var status = document.getElementById('activity-filter-status');
    var currentYear = new Date().getFullYear();

    years.forEach(function (year) {
      var yearNumber = parseInt(year.getAttribute('data-year-section'), 10);
      year.open = yearNumber === currentYear;
    });
    
    function entryMatches(entry, filter) {
      if (filter === 'all') return true;
      if (filter === 'slides') return entry.getAttribute('data-has-slides') === 'true';
      return entry.getAttribute('data-type') === filter;
    }
    
    function updateButtonCounts() {
      buttons.forEach(function (button) {
        var filter = button.getAttribute('data-activity-filter');
        var count = entries.filter(function (entry) { return entryMatches(entry, filter); }).length;
        button.querySelector('[data-filter-count]').textContent = '(' + count + ')';
      });
    }
    
    function applyFilter(filter) {
      var visibleTotal = 0;
    
      entries.forEach(function (entry) {
        var visible = entryMatches(entry, filter);
        entry.hidden = !visible;
        if (visible) visibleTotal += 1;
      });
    
      years.forEach(function (year) {
        var visibleEntries = Array.prototype.slice.call(year.querySelectorAll('.activity-entry')).filter(function (entry) {
          return !entry.hidden;
        });
        year.hidden = visibleEntries.length === 0;
        year.querySelector('[data-visible-year-count]').textContent = visibleEntries.length;
        if (filter === 'all') {
          year.open = parseInt(year.getAttribute('data-year-section'), 10) === currentYear;
        } else {
          year.open = visibleEntries.length > 0;
        }
        var yearLink = document.querySelector('[data-year-link="' + year.getAttribute('data-year-section') + '"]');
        if (yearLink) yearLink.hidden = year.hidden;
      });
    
      buttons.forEach(function (button) {
        button.setAttribute('aria-pressed', button.getAttribute('data-activity-filter') === filter ? 'true' : 'false');
      });
    
      status.textContent = '显示 ' + visibleTotal + ' 项，共 ' + entries.length + ' 项。';
    }
    
    function revealHashTarget() {
      if (!window.location.hash) return;
      var target = document.getElementById(window.location.hash.slice(1));
      if (!target) return;
      var year = target.closest('.activity-year');
      if (year) year.open = true;
    }
    
    buttons.forEach(function (button) {
      button.addEventListener('click', function () {
        applyFilter(button.getAttribute('data-activity-filter'));
      });
    });

    yearActionButtons.forEach(function (button) {
      button.addEventListener('click', function () {
        var shouldExpand = button.getAttribute('data-year-action') === 'expand';
        years.forEach(function (year) {
          if (!year.hidden) year.open = shouldExpand;
        });
      });
    });
    
    Array.prototype.slice.call(document.querySelectorAll('.activity-highlight-card, .activity-entry-links a[href^="#"]')).forEach(function (link) {
      link.addEventListener('click', function () {
        var target = document.getElementById(link.getAttribute('href').slice(1));
        if (!target) return;
        applyFilter('all');
        var year = target.closest('.activity-year');
        if (year) year.open = true;
      });
    });
    
    window.addEventListener('hashchange', revealHashTarget);
    updateButtonCounts();
    applyFilter('all');
    revealHashTarget();
  }());
</script>
